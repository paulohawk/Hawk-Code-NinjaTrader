const SHEET_TRADES    = "Trades";

const TRADES_HEADER_ROW = 1;
const TRADES_DATA_START_ROW = 2;
const TRADES_COLUMNS = 4; // Date, PnL, Contracts, Asset

/* ================= MENU ================= */
function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("DashFlix")
    .addItem("🔄 Atualizar Dashboard", "updateDashboard")
    .addSeparator()
    .addItem("🛠️ Setup Inicial / Reset Layout", "setupDashflix")
    .addToUi();
}

/* ================= EVENT ================= */
function onEdit(e) {
  if (!e || !e.range) return;

  const sh = e.range.getSheet();
  const name = sh.getName();

  // Atualiza automaticamente se mudar o FILTRO no Dashboard.
  if (name === SHEET_DASHBOARD) {
    const a1 = e.range.getA1Notation();
    const r = e.range.getRow();
    const c = e.range.getColumn();
    
    // Verifica se a edição foi na área do seletor de período (M1:P3)
    // M=13, P=16
    if ((a1 === PERIOD_CELL) || (r <= 3 && c >= 13 && c <= 16)) {
      updateDashboard();
    }
  }

  // Atualiza quando a edição acontece na aba Trades (dados relevantes).
  if (name === SHEET_TRADES) {
    const r = e.range.getRow();
    const c = e.range.getColumn();
    if (r >= TRADES_DATA_START_ROW && c >= 1 && c <= TRADES_COLUMNS) {
      updateDashboard();
    }
  }
}

/* ================= SETUP ================= */
function setupDashflix(opts) {
  opts = opts || {};
  const ss = SpreadsheetApp.getActive();

  const shTrades = ensureTradesSheet_(ss);
  const shDash   = ensureDashboardSheet_(ss);

  // Layout do dashboard (Pesado - só roda no Setup)
  ensureDashboardLayout_(shDash);
  ensureCharts_(shDash);

  // Mantém Trades formatado (tema + validações em TODA a planilha)
  formatTradesSheet_(shTrades);

  if (!opts.silent) {
    SpreadsheetApp.getUi().alert("Setup concluído! O layout foi redefinido e as validações aplicadas.");
    updateDashboard(); // Já puxa os dados iniciais
  }
}

/* ================= ENTRY ================= */
function updateDashboard() {
  const ss = SpreadsheetApp.getActive();
  const shDash = ss.getSheetByName(SHEET_DASHBOARD);
  if (!shDash) return; // Se não existir, não faz nada (usuário deve rodar setup)

  // REMOVIDO: ensureDashboardLayout_() 
  // Motivo: Performance. Não recriamos o fundo preto a cada clique. 
  // Se o layout quebrar, o usuário usa o menu "Setup".

  const period = getSelectedPeriod_(shDash);
  const trades = getFilteredTrades_(ss, period);
  const dashboardData = buildDashboardData_(trades);

  // Renderização (Atualiza apenas valores e gráficos)
  renderKpis_(shDash, trades, dashboardData);
  ensureCharts_(shDash);
  renderCharts_(shDash, dashboardData);
  renderRightColumn_(shDash, trades);

  SpreadsheetApp.flush();
}

/* ================= PERIOD ================= */
function getSelectedPeriod_(shDash) {
  let val = shDash.getRange(PERIOD_CELL).getValue();
  val = String(val || "").trim();
  if (!val) val = "Diário";
  return val;
}

/* ================= DATA (Trades only) ================= */
function getFilteredTrades_(ss, period) {
  const sh = ss.getSheetByName(SHEET_TRADES);
  if (!sh) return [];

  const lastRow = sh.getLastRow();
  if (lastRow < TRADES_DATA_START_ROW) return [];

  // Pega todos os dados de uma vez
  const data = sh.getRange(TRADES_DATA_START_ROW, 1, lastRow - TRADES_HEADER_ROW + 1, TRADES_COLUMNS).getValues();

  // Lê e normaliza com try/catch para robustez
  const rows = [];
  for (let i = 0; i < data.length; i++) {
    const r = data[i];
    try {
      // Ignora linhas sem data válida
      if (!(r[0] instanceof Date) || isNaN(r[0].getTime())) continue;
      
      const pnlVal = toNumber_(r[1]);
      const asset = String(r[3] || "").trim();
      
      // Se não tiver ativo ou PnL não for número, pula (para não sujar gráficos)
      if (isNaN(pnlVal) || !asset) continue;

      rows.push({
        date: normalizeDate_(r[0]),
        pnl:  pnlVal,
        contracts: toNumber_(r[2]),
        asset
      });
    } catch (e) {
      // Linha corrompida ignorada silenciosamente
      continue;
    }
  }

  const p = String(period || "").trim().toLowerCase();
  if (!p || p === "geral") return rows;

  const now = normalizeDate_(new Date());

  return rows.filter(t => {
    if (p === "diário" || p === "diario") return sameDay_(t.date, now);
    if (p === "semanal") return sameWeekMon_(t.date, now);
    if (p === "mensal") return (t.date.getMonth() === now.getMonth() && t.date.getFullYear() === now.getFullYear());
    if (p === "anual")  return (t.date.getFullYear() === now.getFullYear());
    return true;
  });
}

function normalizeDate_(d) {
  const x = new Date(d);
  x.setHours(0, 0, 0, 0);
  return x;
}

function toNumber_(v) {
  if (v === "" || v === null || v === undefined) return NaN;
  if (typeof v === "number") return v;
  // Tenta limpar strings como "R$ 1.000,00" ou "1,000.00"
  try {
    const s = String(v).replace(/[^\d.,-]/g, "").replace(",", "."); 
    const n = parseFloat(s);
    return isNaN(n) ? NaN : n;
  } catch (e) {
    return NaN;
  }
}

function sameDay_(a, b) {
  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate();
}

function sameWeekMon_(a, b) {
  return startOfWeekMon_(a).getTime() === startOfWeekMon_(b).getTime();
}

function startOfWeekMon_(date) {
  const d = new Date(date);
  d.setHours(0,0,0,0);
  const day = (d.getDay() + 6) % 7; // Mon=0
  d.setDate(d.getDate() - day);
  return d;
}

/* ================= KPI RENDER ================= */
function renderKpis_(sh, trades, dashboardData) {
  const totalPnL = sum_(trades.map(t => t.pnl));
  const daysCount = countUnique_(trades.map(t => t.date.getTime()));
  const wins = trades.filter(t => t.pnl > 0).length;
  const winRate = trades.length ? Math.round((wins / trades.length) * 100) : 0;

  const efficiencyRows = (dashboardData && dashboardData.efficiency) ? dashboardData.efficiency : buildEfficiency_(trades);
  const pos = Number((efficiencyRows[1] || [0, 0])[1]) || 0;
  const negAbs = Number((efficiencyRows[2] || [0, 0])[1]) || 0;
  const efficiency = (pos + negAbs) > 0 ? Math.round((pos / (pos + negAbs)) * 100) : 0;

  const daily = (dashboardData && dashboardData.daily) ? dashboardData.daily : aggregateDaily_(trades);
  const cumulativeRows = (dashboardData && dashboardData.cumulative) ? dashboardData.cumulative : buildCumulative_(daily);
  const cumArr = cumulativeRows.slice(1).map(row => Number(row[1]) || 0);
  const maxDD = computeMaxDrawdown_(cumArr);

  const pnlMedio = trades.length ? (totalPnL / trades.length) : 0;

  // Valores KPI (Linha 5)
  // 1,3,5,7,9,11
  const vals = [
    formatUSD_(totalPnL),
    efficiency + "%",
    daysCount,
    winRate + "%",
    formatUSD_(maxDD),
    formatUSD_(pnlMedio)
  ];

  // Escreve em lote para performance (uma chamada por célula infelizmente devido ao merge, mas ok)
  sh.getRange(5, 1).setValue(vals[0]);
  sh.getRange(5, 3).setValue(vals[1]);
  sh.getRange(5, 5).setValue(vals[2]);
  sh.getRange(5, 7).setValue(vals[3]);
  sh.getRange(5, 9).setValue(vals[4]);
  sh.getRange(5, 11).setValue(vals[5]);
}

function sum_(arr){ return (arr || []).reduce((s,v)=>s+(Number(v)||0),0); }
function countUnique_(arr){ return new Set(arr || []).size; }
function formatUSD_(n){ return "$ " + Number(n||0).toFixed(2); }
function computeMaxDrawdown_(cumArr){
  let peak = -Infinity;
  let maxDD = 0;
  for (let i=0;i<cumArr.length;i++){
    const v = Number(cumArr[i]||0);
    if (v > peak) peak = v;
    const dd = peak - v;
    if (dd > maxDD) maxDD = dd;
  }
  return maxDD; // Retorna positivo para exibição
}

/* ================= CHARTS ================= */
function renderCharts_(sh, dashboardData) {
  const daily = (dashboardData && dashboardData.daily) ? dashboardData.daily : [];
  const curve = (dashboardData && dashboardData.cumulative) ? dashboardData.cumulative : [["Data", "PnL"]];
  const eff   = (dashboardData && dashboardData.efficiency) ? dashboardData.efficiency : [["Tipo", "Valor"]];
  const bars  = daily.length ? buildBars_(daily) : [["Data", "PnL"]];

  const curveRange = sh.getRange(CHART_DATA_START_ROW, 1, CHART_CURVE_ROWS, 2);
  const effRange   = sh.getRange(CHART_DATA_START_ROW, 4, CHART_EFF_ROWS, 2);
  const barsRange  = sh.getRange(CHART_DATA_START_ROW, 7, CHART_BARS_ROWS, 2);

  curveRange.clearContent();
  effRange.clearContent();
  barsRange.clearContent();

  curveRange.setValues(padTable_(curve, CHART_CURVE_ROWS, 2));
  effRange.setValues(padTable_(eff, CHART_EFF_ROWS, 2));
  barsRange.setValues(padTable_(bars, CHART_BARS_ROWS, 2));

  updateChartsAppearance_(sh);
}

function buildDashboardData_(trades) {
  const daily = aggregateDaily_(trades);
  const cumulative = daily.length ? buildCumulative_(daily) : [["Data", "PnL"]];
  const efficiency = buildEfficiency_(trades);
  return { daily, cumulative, efficiency };
}

function aggregateDaily_(trades) {
  const m = new Map();
  trades.forEach(t => {
    const key = Utilities.formatDate(t.date, Session.getScriptTimeZone(), "yyyy-MM-dd");
    const cur = m.get(key) || { date: t.date, pnl: 0 };
    cur.pnl += (Number(t.pnl) || 0);
    m.set(key, cur);
  });
  return Array.from(m.values()).sort((a,b)=>a.date-b.date);
}

function buildCumulative_(daily) {
  const out = [["Data", "PnL"]];
  let cum = 0;
  daily.forEach(d => {
    cum += d.pnl;
    out.push([d.date, cum]);
  });
  return out;
}

function buildBars_(daily) {
  const out = [["Data", "PnL"]];
  daily.forEach(d => out.push([d.date, d.pnl]));
  return out;
}

function buildEfficiency_(trades) {
  const pos = sum_(trades.filter(t => t.pnl > 0).map(t => t.pnl));
  const neg = sum_(trades.filter(t => t.pnl < 0).map(t => Math.abs(t.pnl)));
  return [ ["Tipo", "Valor"], ["Positivo", pos], ["Negativo", neg] ];
}

function ensureCharts_(sh) {
  ensureChartDataRange_(sh);

  const charts = sh.getCharts();
  if (charts.length >= 3) return;

  const commonOpts = {
    backgroundColor: { fill: THEME.bg },
    chartArea: { backgroundColor: THEME.bg },
    hAxis: { textStyle: { color: THEME.text }, gridlines: { color: THEME.grid } },
    vAxis: { textStyle: { color: THEME.text }, gridlines: { color: THEME.grid } },
    legend: { position: "none" },
    width: CHART_WIDTH,
    height: CHART_HEIGHT
  };

  const curveRange = sh.getRange(CHART_DATA_START_ROW, 1, CHART_CURVE_ROWS, 2);
  const effRange   = sh.getRange(CHART_DATA_START_ROW, 4, CHART_EFF_ROWS, 2);
  const barsRange  = sh.getRange(CHART_DATA_START_ROW, 7, CHART_BARS_ROWS, 2);

  const c1 = sh.newChart().setChartType(Charts.ChartType.LINE)
    .addRange(curveRange)
    .setPosition(8, 1, 0, 0)
    .setOption("colors", [THEME.bronze]);
  for (let k in commonOpts) c1.setOption(k, commonOpts[k]);
  sh.insertChart(c1.build());

  const c2 = sh.newChart().setChartType(Charts.ChartType.PIE)
    .addRange(effRange)
    .setPosition(21, 1, 0, 0)
    .setOption("pieHole", 0.5)
    .setOption("colors", [THEME.bronze, "#555555"])
    .setOption("legend", { position: "right", textStyle: { color: THEME.text } })
    .setOption("pieSliceText", "percentage")
    .setOption("backgroundColor.fill", THEME.bg)
    .setOption("width", CHART_WIDTH)
    .setOption("height", CHART_HEIGHT)
    .build();
  sh.insertChart(c2);

  const c3 = sh.newChart().setChartType(Charts.ChartType.COLUMN)
    .addRange(barsRange)
    .setPosition(34, 1, 0, 0)
    .setOption("colors", [THEME.bronze]);
  for (let k in commonOpts) c3.setOption(k, commonOpts[k]);
  sh.insertChart(c3.build());
}

function updateChartsAppearance_(sh) {
  const charts = sh.getCharts();
  if (!charts.length) return;

  charts.forEach(chart => {
    const builder = chart.modify();
    builder
      .setOption("backgroundColor", { fill: THEME.bg })
      .setOption("chartArea", { backgroundColor: THEME.bg })
      .setOption("hAxis", { textStyle: { color: THEME.text }, gridlines: { color: THEME.grid } })
      .setOption("vAxis", { textStyle: { color: THEME.text }, gridlines: { color: THEME.grid } })
      .setOption("width", CHART_WIDTH)
      .setOption("height", CHART_HEIGHT);
    sh.updateChart(builder.build());
  });
}

function ensureChartDataRange_(sh) {
  const requiredLastRow = CHART_DATA_START_ROW + Math.max(CHART_CURVE_ROWS, CHART_BARS_ROWS, CHART_EFF_ROWS);
  if (sh.getMaxRows() < requiredLastRow) {
    sh.insertRowsAfter(sh.getMaxRows(), requiredLastRow - sh.getMaxRows());
  }

  sh.getRange(CHART_DATA_START_ROW + 1, 1, CHART_CURVE_ROWS - 1, 1).setNumberFormat("dd/MM/yyyy");
  sh.getRange(CHART_DATA_START_ROW + 1, 7, CHART_BARS_ROWS - 1, 1).setNumberFormat("dd/MM/yyyy");
}

function padTable_(rows, targetRows, targetCols) {
  const out = [];
  for (let r = 0; r < targetRows; r++) {
    const source = rows[r] || [];
    const row = [];
    for (let c = 0; c < targetCols; c++) {
      row.push(source[c] !== undefined ? source[c] : "");
    }
    out.push(row);
  }
  return out;
}

/* ================= RIGHT COLUMN (M:P) ================= */
function renderRightColumn_(sh, trades) {
  const startCol = 13; // M
  const startRow = 6;
  const totalRow = 44;

  const props = PropertiesService.getDocumentProperties();
  const prevCount = parseInt(props.getProperty(RIGHT_COLUMN_ROWS_PROP) || "0", 10);
  const maxRows = sh.getMaxRows();
  if (prevCount > 0 && maxRows > startRow) {
    const rowsToClear = Math.min(prevCount, maxRows - startRow);
    if (rowsToClear > 0) {
      const range = sh.getRange(startRow + 1, startCol, rowsToClear, 4);
      range.clearContent();
    }
  }

  const sorted = trades.slice().sort((a,b)=>b.date - a.date);
  const listStartRow = startRow + 1;
  const maxListRow = totalRow - 1;
  const maxListLength = Math.max(0, maxListRow - listStartRow + 1);
  const limit = Math.min(getRightColumnLimit_(sh), maxListLength);
  const limited = sorted.slice(0, limit);

  if (maxListLength > 0) {
    const listRange = sh.getRange(listStartRow, startCol, maxListLength, 4);
    listRange
      .setFontFamily("Arial")
      .setFontSize(10)
      .setVerticalAlignment("middle");
    sh.setRowHeights(listStartRow, maxListLength, 20);

    const dateRange = sh.getRange(listStartRow, startCol, maxListLength, 1);
    const pnlRange = sh.getRange(listStartRow, startCol + 1, maxListLength, 1);
    const contractsRange = sh.getRange(listStartRow, startCol + 2, maxListLength, 1);
    const assetRange = sh.getRange(listStartRow, startCol + 3, maxListLength, 1);

    dateRange.setNumberFormat("dd/MM/yyyy").setHorizontalAlignment("left");
    pnlRange.setNumberFormat("$#,##0.00").setHorizontalAlignment("right");
    contractsRange.setHorizontalAlignment("center");
    assetRange.setHorizontalAlignment("center");
  }

  sh.setRowHeight(totalRow, 20);
  sh.setRowHeight(totalRow + 1, 22);

  if (limited.length) {
    const rows = limited.map(t => ([
      Utilities.formatDate(t.date, Session.getScriptTimeZone(), "dd/MM/yyyy"),
      Number(t.pnl)||0,
      Number(t.contracts)||"",
      String(t.asset||"")
    ]));

    sh.getRange(listStartRow, startCol, rows.length, 4).setValues(rows);
  }

  // TOTAL
  const total = sum_(trades.map(t => t.pnl));

  // Garante que temos linhas suficientes
  if (totalRow + 2 > sh.getMaxRows()) {
    sh.insertRowsAfter(sh.getMaxRows(), 5);
  }

  const totalLabel = sh.getRange(totalRow, startCol, 1, 4);
  safeMerge_(totalLabel);
  totalLabel.setValue("TOTAL")
    .setBackground(THEME.black)
    .setFontColor(THEME.text)
    .setFontWeight("bold")
    .setHorizontalAlignment("left")
    .setFontSize(12);

  const totalVal = sh.getRange(totalRow + 1, startCol, 1, 4);
  safeMerge_(totalVal);
  totalVal.setValue(formatUSD_(total))
    .setBackground(THEME.black)
    .setFontColor(THEME.bronze)
    .setFontWeight("bold")
    .setHorizontalAlignment("right")
    .setFontSize(14);

  const renderedRows = totalRow - startRow + 1;
  props.setProperty(RIGHT_COLUMN_ROWS_PROP, String(renderedRows));
}

function getRightColumnLimit_(sh) {
  if (!sh) return RIGHT_COLUMN_LIMIT;
  const raw = sh.getRange(RIGHT_COLUMN_LIMIT_CELL).getValue();
  const parsed = parseInt(raw, 10);
  if (!isNaN(parsed) && parsed > 0) return parsed;
  return RIGHT_COLUMN_LIMIT;
}

/* ================= TRADES SHEET (REFATORADO) ================= */
function formatTradesSheet_(sh) {
  sh.setHiddenGridlines(true);

  // Formata somente até a última linha usada + buffer
  const lastUsedRow = Math.max(sh.getLastRow(), TRADES_HEADER_ROW);
  const formatLastRow = lastUsedRow + TRADES_FORMAT_BUFFER;

  const currentMax = sh.getMaxRows();
  if (currentMax < formatLastRow) {
    sh.insertRowsAfter(currentMax, formatLastRow - currentMax);
  }

  const maxRows = Math.min(sh.getMaxRows(), formatLastRow);

  // Formata apenas a coluna A (reduz custo)
  sh.getRange(1, 1, maxRows, 1)
    .setBackground(THEME.bg)
    .setFontColor(THEME.text)
    .setFontFamily("Arial");

  // Header
  sh.getRange(1,1,1,4)
    .setValues([["Data","PnL (USD)","Contratos","Ativo"]])
    .setBackground(THEME.black)
    .setFontColor(THEME.text)
    .setFontWeight("bold")
    .setHorizontalAlignment("center");
  sh.setFrozenRows(1);

  // Larguras
  sh.setColumnWidth(1, 120);
  sh.setColumnWidth(2, 120);
  sh.setColumnWidth(3, 110);
  sh.setColumnWidth(4, 110);

  // Formatações de Coluna no intervalo reduzido
  const rowsToFormat = Math.max(maxRows - 1, 0);
  if (rowsToFormat > 0) {
    // Formatos numéricos
    sh.getRange(2,1,rowsToFormat,1).setNumberFormat("dd/MM/yyyy");
    sh.getRange(2,2,rowsToFormat,1).setNumberFormat("$#,##0.00");
    
    // Alinhamentos
    sh.getRange(2,3,rowsToFormat,2).setHorizontalAlignment("center");

    // Validações
    const dateRule = SpreadsheetApp.newDataValidation().requireDate().setAllowInvalid(false).build();
    const pnlRule = SpreadsheetApp.newDataValidation().requireNumberBetween(-1e9, 1e9).setAllowInvalid(false).build();
    const ctrRule = SpreadsheetApp.newDataValidation().requireNumberGreaterThan(0).setAllowInvalid(false).build();
    const assetRule = SpreadsheetApp.newDataValidation().requireValueInList(DEFAULT_ASSETS, true).setAllowInvalid(false).build();

    sh.getRange(2,1,rowsToFormat,1).setDataValidation(dateRule);
    sh.getRange(2,2,rowsToFormat,1).setDataValidation(pnlRule);
    sh.getRange(2,3,rowsToFormat,1).setDataValidation(ctrRule);
    sh.getRange(2,4,rowsToFormat,1).setDataValidation(assetRule);
  }
}

/* ================= UTILS ================= */
function safeMerge_(range) {
  try { range.breakApart(); } catch (e) {}
  range.merge();
}
